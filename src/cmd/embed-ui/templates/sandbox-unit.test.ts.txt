import { describe, expect, test } from 'vitest'

import extensionrc from '../../../../extensionrc.json'

describe('extension manifest', () => {
  test('contains generated page and widget target', () => {
    expect(extensionrc.pages).toEqual(
      expect.arrayContaining([
        expect.objectContaining({ code: '__PAGE_CODE__' }),
      ])
    )

    expect(extensionrc.targets).toContain('__WIDGET_TARGET__')
    expect(extensionrc.runner).toBe('worker')
    expect(extensionrc.entrypointType).toBe('script')
  })
})
