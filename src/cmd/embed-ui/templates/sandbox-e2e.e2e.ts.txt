import fs from 'node:fs'
import path from 'node:path'

import { expect, test } from '@playwright/test'

import {
  createSandboxPagePath,
  readSandboxSnapshot,
} from '@retailcrm/embed-ui-v1-sandbox/automation/playwright'

type ExtensionConfig = {
  pages: Array<{ code: string }>
  targets: string[]
  uuid: string
}

const extensionBaseURL = process.env.SANDBOX_EXTENSION_URL?.trim()
const extensionrc = JSON.parse(
  fs.readFileSync(path.resolve(process.cwd(), 'extensionrc.json'), 'utf8'),
) as ExtensionConfig
const extensionUrl = extensionBaseURL
  ? new URL(extensionrc.uuid, extensionBaseURL).href
  : ''
const [pageDescriptor] = extensionrc.pages
const [target] = extensionrc.targets
const missingExtensionUrlMessage = 'Fill SANDBOX_EXTENSION_URL in .env.sandbox before running starter.e2e.ts.'

if (!extensionBaseURL) {
  console.warn(`[sandbox:e2e] ${missingExtensionUrlMessage}`)
}

test.skip(
  !extensionBaseURL,
  missingExtensionUrlMessage,
)

test('loads generated settings page extension and saves settings', async ({ page }) => {
  await page.goto(createSandboxPagePath({
    extensionUrl,
    manifestUrl: extensionUrl,
    pageCode: pageDescriptor.code,
    targets: [target],
  }))

  await expect(page).toHaveURL(/mode=page/u)
  await expect(page).toHaveURL(new RegExp(`pageCode=${pageDescriptor.code}`, 'u'))
  await expect(page.getByRole('heading', { name: 'Настройки расширения' })).toBeVisible()
  await expect(page.locator('svg[class*="settings-page-addon__icon"]')).toHaveCSS('width', '24px')
  await expect(page.locator('svg[class*="settings-page-addon__icon"]')).toHaveCSS('height', '24px')
  await expect(page.getByRole('button', { name: 'Сохранить' })).toBeVisible()
  await expect(page.getByLabel('Название помощника')).toHaveValue('Order follow-up')

  await page.getByLabel('Название помощника').fill('E2E assistant')
  await page.getByRole('button', { name: 'Сохранить' }).click()

  await expect(page.locator('body')).toContainText('Настройки сохранены')

  const snapshot = await readSandboxSnapshot(page)

  expect(snapshot.host.http.every(record => record.response.status === 200)).toBe(true)
})
