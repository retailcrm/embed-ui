import fs from 'node:fs'
import path from 'node:path'

import { expect, test } from '@playwright/test'

import {
  launchSandboxExtension,
} from '@retailcrm/embed-ui-v1-sandbox/automation/playwright'
import {
  parseSandboxExtensionDescriptorJson,
} from '@retailcrm/embed-ui-v1-sandbox/scenario'

type ExtensionConfig = {
  pages: Array<{ code: string }>
  runner: string
  stylesheet: boolean
  targets: string[]
  uuid: string
}

const extensionBaseURL = process.env.SANDBOX_EXTENSION_URL?.trim()
  || 'http://127.0.0.1:4175/extension/'
const extensionrc = JSON.parse(
  fs.readFileSync(path.resolve(process.cwd(), 'extensionrc.json'), 'utf8'),
) as ExtensionConfig
const extensionUrl = new URL(`${extensionrc.uuid}/`, extensionBaseURL)
const descriptor = parseSandboxExtensionDescriptorJson(JSON.stringify({
  runner: extensionrc.runner,
  entrypoint: new URL('script', extensionUrl).href,
  stylesheet: extensionrc.stylesheet ? new URL('stylesheet', extensionUrl).href : null,
  pages: extensionrc.pages.map(page => page.code),
  targets: extensionrc.targets,
}))
const [pageCode] = descriptor.pages

test('loads generated settings page extension and saves settings', async ({ page }) => {
  if (!descriptor.stylesheet) throw new Error('The settings page requires a stylesheet.')

  const scriptResponse = page.waitForResponse(descriptor.entrypoint)
  const stylesheetResponse = page.waitForResponse(descriptor.stylesheet)

  await page.goto('/')
  await launchSandboxExtension(page, {
    mode: 'page',
    descriptor,
    pageCode,
  })

  expect((await scriptResponse).status()).toBe(200)
  expect((await stylesheetResponse).status()).toBe(200)

  await expect(page).toHaveURL(url => url.search === '')
  await expect(page.getByRole('heading', { name: 'Настройки расширения' })).toBeVisible()
  const assistantName = page.getByLabel('Название помощника')
  const note = page.getByLabel('Заметка по умолчанию')
  const enabled = page.getByRole('switch', { name: 'Включить подсказки в заказе' })

  await expect(assistantName).toHaveValue('Order follow-up')
  await expect(note).toHaveValue('Check payment status and delivery date before calling the customer.')
  await expect(enabled).toHaveAttribute('aria-checked', 'true')

  await assistantName.fill('E2E assistant')
  await note.fill('Call the customer after payment confirmation.')
  await enabled.click()
  await page.getByRole('button', { name: 'Сохранить' }).click()

  await expect(page.getByText('Настройки сохранены')).toBeVisible()
  await expect(assistantName).toHaveValue('E2E assistant')
  await expect(note).toHaveValue('Call the customer after payment confirmation.')
  await expect(enabled).toHaveAttribute('aria-checked', 'false')
})
